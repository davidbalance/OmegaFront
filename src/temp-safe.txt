const biologicals: biologicalFormType[] = ["biological.virus", "biological.fungi", "biological.bacteria", "biological.parasites", "biological.vectorExposure", "biological.wildAnimalExposure"];
const ergonomics: ergonomicFormType[] = ["ergonomic.manualHandling", "ergonomic.repetitiveMovements", "ergonomic.forcedPostures", "ergonomic.pvdWork", "ergonomic.poorWorkstationDesign"];
const psychosocials: psychosocialFormType[] = ["psychosocial.monotony", "psychosocial.workOverload", "psychosocial.taskDetail", "psychosocial.highResponsibility", "psychosocial.decisionAutonomy", "psychosocial.poorSupervision", "psychosocial.roleConflict", "psychosocial.unclearResponsibilities", "psychosocial.poorTaskDistribution", "psychosocial.rotatingShifts", "psychosocial.interpersonalRelations", "psychosocial.jobInstability", "psychosocial.criminalThreat"];

const safetyOptions = [locatives, mechanicals, electricals];
const maxSafetyRows = safetyOptions.reduce((acc, prev) => acc < prev.length ? prev.length : acc, 0)

const MAX_JOB_RISKS_LENGTH: number = 7;

const JOB_RISK_OPTION_PHYSICAL = "physical"
const JOB_RISK_OPTION_SAFETY = "locative"
const JOB_RISK_OPTION_CHEMICAL = "chemical"
const JOB_RISK_OPTION_BIOLOGICAL = "biological"
const JOB_RISK_OPTION_ERGONOMIC = "ergonomic"
const JOB_RISK_OPTION_PSYCHOSOCIAL = "psychosocial"
const jobRisksOptions: Option[] = [
    { label: "Físico", value: JOB_RISK_OPTION_PHYSICAL },
    { label: "De Seguridad", value: JOB_RISK_OPTION_SAFETY },
    { label: "Químico", value: JOB_RISK_OPTION_CHEMICAL },
    { label: "Biológico", value: JOB_RISK_OPTION_BIOLOGICAL },
    { label: "Ergonómico", value: JOB_RISK_OPTION_ERGONOMIC },
    { label: "PsicoSocial", value: JOB_RISK_OPTION_PSYCHOSOCIAL }
]

type FemoJobRiskFormProps = {
    data?: Partial<JobRisksSchemaType>;
    onSubmit?: (value: JobRisksSchemaType) => void;
}
const FemoJobRiskForm = React.forwardRef<HTMLFormElement, FemoJobRiskFormProps>(({
    data,
    onSubmit
}, ref) => {

    const [jobRiskType, setJobRiskType] = useState<string>("")

    const { onSubmit: formSubmit, getInputProps, values: formValues, setValues } = useForm<JobRisksSchemaType>({
        initialValues: adjustInitialValue(data),
        validate: zodResolver(JobRiskSchema)
    });

    const jobRisks: JobRiskSchemaType[] = formValues.jobRisks ?? [];

    const handleSubmit = useCallback((value: JobRisksSchemaType) => {
        console.log(value);
        // onSubmit?.(value);
    }, [onSubmit]);

    const handleAdd = () => setValues(prev => (prev.jobRisks?.length ?? 0) < MAX_JOB_RISKS_LENGTH ? ({
        ...prev,
        jobRisks: [...(prev.jobRisks ?? []), DEFAULT_JOB_RISK]
    }) : prev);

    const handleRemove = (index: number) => setValues(prev => ({
        ...prev,
        jobRisks: [...(prev.jobRisks?.slice(0, index) ?? []), ...(prev.jobRisks?.slice(index + 1) ?? [])]
    }));

    const physicalForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh>Físico</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {physicals.map((examKey, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}-${examKey}`}>
                            <TableTd>
                                <Checkbox
                                    label={jobRisksFormLabels[examKey]}
                                    {...getInputProps(`jobRisks.${index}.${examKey}`)} />
                            </TableTd>
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.physical.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const safetyForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh colSpan={3}>De Seguridad</TableTh>
                    </TableTr>
                    <TableTr>
                        <TableTh>Locativos</TableTh>
                        <TableTh>Mecánicos</TableTh>
                        <TableTh>Electricos</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {Array.from({ length: maxSafetyRows }).map((_, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}`}>
                            {safetyOptions.map((e, jobRisk) => {
                                const key = e[subIndex];
                                if (key) {
                                    return (
                                        <TableTd key={`${index}-${subIndex}-${jobRisk}-${key}`}>
                                            <Checkbox
                                                label={jobRisksFormLabels[key]}
                                                {...getInputProps(`jobRisks.${index}.${key}`)} />
                                        </TableTd>
                                    )
                                } else {
                                    return (
                                        <TableTd key={`${index}-${subIndex}-${jobRisk}-empty`}>{" "}</TableTd>
                                    )
                                }
                            })}
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd colSpan={3}>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.safety.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const chemicalForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh>Químico</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {chemicals.map((examKey, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}-${examKey}`}>
                            <TableTd>
                                <Checkbox
                                    label={jobRisksFormLabels[examKey]}
                                    {...getInputProps(`jobRisks.${index}.${examKey}`)} />
                            </TableTd>
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.chemical.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const biologicalForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh>Biológico</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {biologicals.map((examKey, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}-${examKey}`}>
                            <TableTd>
                                <Checkbox
                                    label={jobRisksFormLabels[examKey]}
                                    {...getInputProps(`jobRisks.${index}.${examKey}`)} />
                            </TableTd>
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.biological.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const ergonomicForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh>Ergonómico</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {ergonomics.map((examKey, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}-${examKey}`}>
                            <TableTd>
                                <Checkbox
                                    label={jobRisksFormLabels[examKey]}
                                    {...getInputProps(`jobRisks.${index}.${examKey}`)} />
                            </TableTd>
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.ergonomic.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const psychosocialForm = (index: number) => {
        return (
            <>
                <TableThead>
                    <TableTr>
                        <TableTh>PsicoSocial</TableTh>
                    </TableTr>
                </TableThead>
                <TableTbody>
                    {psychosocials.map((examKey, subIndex) => (
                        <TableTr key={`form-key-${index}-${subIndex}-${examKey}`}>
                            <TableTd>
                                <Checkbox
                                    label={jobRisksFormLabels[examKey]}
                                    {...getInputProps(`jobRisks.${index}.${examKey}`)} />
                            </TableTd>
                        </TableTr>
                    ))}
                    <TableTr>
                        <TableTd>
                            <TextInput
                                label="Otros"
                                {...getInputProps(`jobRisks.${index}.psychosocial.other`)} />
                        </TableTd>
                    </TableTr>
                </TableTbody>
            </>
        )
    }

    const formMapper = useCallback((index: number) => {
        switch (jobRiskType) {
            case JOB_RISK_OPTION_SAFETY:
                return safetyForm(index)
            case JOB_RISK_OPTION_CHEMICAL:
                return chemicalForm(index)
            case JOB_RISK_OPTION_BIOLOGICAL:
                return biologicalForm(index)
            case JOB_RISK_OPTION_ERGONOMIC:
                return ergonomicForm(index)
            case JOB_RISK_OPTION_PSYCHOSOCIAL:
                return psychosocialForm(index)
            case JOB_RISK_OPTION_PHYSICAL:
            default:
                return physicalForm(index)
        }
    }, [jobRiskType])

    return (
        <>
            <Title order={3}>Factores de riesgo del trabajo actual</Title>
            <Box
                mt={rem(16)}
                component='form'
                ref={ref}
                onSubmit={formSubmit(handleSubmit)}>
                <Stack gap={rem(32)}>
                    <Select
                        data={jobRisksOptions}
                        checkIconPosition="left"
                        label="Tipo de riesgo"
                        defaultDropdownOpened={false}
                        maxDropdownHeight={200}
                        value={jobRiskType}
                        onChange={(value) => setJobRiskType(value ?? JOB_RISK_OPTION_PHYSICAL)} />

                    {jobRisks.map((_, i) => (
                        <Stack key={`root-${i}`} gap={rem(16)}>
                            <Flex gap={rem(8)}>
                                {jobRisks.length - 1 === i && jobRisks.length < MAX_JOB_RISKS_LENGTH && (
                                    <ActionIcon
                                        w="100%"
                                        variant='light'
                                        onClick={handleAdd}>
                                        <IconPlus style={{ width: rem(16), height: rem(16) }} />
                                    </ActionIcon>)}
                                {jobRisks.length > 1
                                    && (
                                        <ActionIcon variant='light'
                                            w="100%"
                                            onClick={() => handleRemove(i)}>
                                            <IconMinus style={{ width: rem(16), height: rem(16) }} />
                                        </ActionIcon>)}
                            </Flex>
                            <Stack w="100%">
                                <Table w="100%" withColumnBorders withRowBorders={false}>
                                    {formMapper(i)}
                                </Table>
                                <Textarea
                                    label="Medidas Preventivas"
                                    placeholder='eg. Lorem Ipsum...'
                                    rows={5}
                                    {...getInputProps(`jobRisks.${i}.preventiveMeasures`)} />
                                {(jobRisks.length > 1 && i < jobRisks.length - 1) && <Divider />}
                            </Stack>
                        </Stack>
                    ))}
                </Stack>
            </Box >
        </>
    )
});

FemoJobRiskForm.displayName = 'FemoJobRiskForm';

export default FemoJobRiskForm
